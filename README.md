- Build instructions

`
mkdir BUILD
cd BUILD
cmake .. -DMLIR_DIR=<LLVM_INSTALL_DIR>/lib/cmake/mlir
cmake --build .
`

- LLVM Toolchain Build
`
cmake -G Ninja -DCMAKE_BUILD_TYPE=Release -DLLVM_ENABLE_ASSERTIONS=OFF  -DCMAKE_INSTALL_PREFIX="<LLVM_INSTALL_DIR>" -DLLVM_OPTIMIZED_TABLEGEN=ON -DLLVM_BUILD_TESTS=ON -DLLVM_DEFAULT_TARGET_TRIPLE="x86_64-unknown-linux-gnu" -DLLVM_ENABLE_PROJECTS="clang;mlir;lld"  -DCLANG_ENABLE_CIR=ON -DLLVM_TARGETS_TO_BUILD="X86;RISCV" -DLLVM_USE_LINKER=lld -DLLVM_PARALLEL_LINK_JOBS=1 -DLLVM_INSTALL_UTILS=ON -DBUILD_SHARED_LIBS=OFF -DLLVM_BUILD_LLVM_DYLIB=ON  -DLLVM_LINK_LLVM_DYLIB=ON  -DMLIR_BUILD_MLIR_C_DYLIB=ON <LLVM_SRC_DIR>/llvm-project/llvm
`
